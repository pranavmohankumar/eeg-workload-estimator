4. Core Technical Starter Skeleton

import numpy as np
from scipy.signal import welch

def extract_eeg_features(eeg_data, fs=256):
    # eeg_data shape: (channels, samples)
    features = []
    for ch in range(eeg_data.shape[0]):
        freqs, psd = welch(eeg_data[ch], fs=fs, nperseg=fs)
        theta = np.trapz(psd[(freqs>=4) & (freqs<=7)])
        alpha = np.trapz(psd[(freqs>=8) & (freqs<=12)])
        beta = np.trapz(psd[(freqs>=13) & (freqs<=30)])
        features.extend([theta, alpha, beta, beta / (alpha + 1e-6)])
    return np.array(features)
